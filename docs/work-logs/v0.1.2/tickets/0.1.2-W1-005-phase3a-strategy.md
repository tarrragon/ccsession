### 2.2 Migrations.py 遷移邏輯（虛擬碼）

```pseudocode
def migrate_ticket(ticket_data):
    """
    將 ticket data 從任意版本遷移至最新版本。

    Args:
        ticket_data: dict，ticket frontmatter 資料

    Returns:
        tuple: (migrated_data, migration_history)
        - migrated_data: 遷移後的資料
        - migration_history: 遷移步驟記錄 [{"from": "1.0", "to": "2.0", "status": "success"}]
    """

    # 步驟 1：偵測當前版本
    current_version = ticket_data.get("protocol_version", PROTOCOL_VERSION_DEFAULT)

    # 步驟 2：版本驗證
    if not is_valid_version_format(current_version):
        raise ProtocolVersionError(f"無效版本格式: {current_version}")

    # 步驟 3：初始化遷移歷史
    migration_history = []

    # 步驟 4：逐步遷移至最新版本
    while current_version != PROTOCOL_VERSION_CURRENT:
        if current_version not in PROTOCOL_VERSION_MIGRATIONS:
            raise ProtocolVersionError(f"無遷移路徑：{current_version} → {PROTOCOL_VERSION_CURRENT}")

        # 取得遷移規則
        migration_rule = PROTOCOL_VERSION_MIGRATIONS[current_version]
        target_version = migration_rule["target"]
        handler_name = migration_rule["handler"]

        # 執行遷移
        handler = get_migration_handler(handler_name)
        ticket_data = handler(ticket_data)

        # 更新版本欄位
        ticket_data["protocol_version"] = target_version

        # 記錄遷移步驟
        migration_history.append({
            "from": current_version,
            "to": target_version,
            "status": "success",
            "handler": handler_name,
        })

        current_version = target_version

    # 步驟 5：返回遷移結果和歷史
    return ticket_data, migration_history


def migrate_v1_to_v2(ticket_data):
    """
    v1.0 → v2.0 遷移函式（向後相容）。

    遷移內容：
    - 保留所有現有欄位（無欄位刪除或重定義）
    - 新增可選欄位的預設值（如 creation_accepted, tdd_phase 等）

    Args:
        ticket_data: dict，v1.0 schema 的 ticket 資料

    Returns:
        dict，v2.0 schema 的 ticket 資料
    """

    # 保留所有原有欄位（100% 無資訊遺失）
    migrated_data = ticket_data.copy()

    # 新增 v2.0 的可選欄位預設值
    if "creation_accepted" not in migrated_data:
        migrated_data["creation_accepted"] = False

    if "tdd_phase" not in migrated_data:
        migrated_data["tdd_phase"] = None

    if "tdd_stage" not in migrated_data:
        migrated_data["tdd_stage"] = []

    # v2.0 新增的其他可選欄位...

    return migrated_data


def is_valid_version_format(version_string):
    """驗證版本格式是否為 Major.Minor 形式"""
    return PROTOCOL_VERSION_RE.match(version_string) is not None
```

### 2.3 protocol-version-check-hook 邏輯（虛擬碼）

```pseudocode
hook_type = "PostToolUse"  # Ticket 讀取相關操作後觸發

def protocol_version_check_hook(context):
    """
    在 ticket 載入時自動檢測版本並觸發遷移。

    觸發時機：
    - ticket track 讀取 Ticket
    - 列表 Ticket
    - 驗證 Ticket 時

    Args:
        context: Hook 上下文，包含 ticket_data 或 ticket_path

    Returns:
        dict，Hook 結果 {"status": "success"/"warning", "message": "..."}
    """

    ticket_data = context.get("ticket_data")
    if not ticket_data:
        return {"status": "skipped", "reason": "非 Ticket 資料"}

    # 步驟 1：偵測版本
    detected_version = ticket_data.get("protocol_version", PROTOCOL_VERSION_DEFAULT)

    # 步驟 2：驗證格式
    if not is_valid_version_format(detected_version):
        # 格式異常：輸出 WARNING，不阻止操作
        error_msg = f"Ticket {ticket_data.get('id', '?')}: 版本格式異常 '{detected_version}'，期望格式 'Major.Minor'"

        # 雙通道記錄（stderr + 日誌檔）
        log_to_stderr(level="WARNING", message=error_msg)
        log_to_file(level="WARNING", message=error_msg, logdir=".claude/hook-logs/protocol-version-check/")

        return {
            "status": "warning",
            "message": error_msg,
            "ticket_id": ticket_data.get("id"),
        }

    # 步驟 3：版本比對
    if detected_version == PROTOCOL_VERSION_CURRENT:
        # 版本已是最新
        log_to_file(level="INFO", message=f"Ticket {ticket_data.get('id')}: 版本已最新 {detected_version}")
        return {"status": "success", "reason": "version_already_current"}

    # 步驟 4：觸發遷移
    if detected_version not in PROTOCOL_VERSIONS_SUPPORTED:
        error_msg = f"Ticket {ticket_data.get('id')}: 版本不支援 '{detected_version}'（支援版本：{PROTOCOL_VERSIONS_SUPPORTED}）"
        log_to_stderr(level="WARNING", message=error_msg)
        log_to_file(level="WARNING", message=error_msg)

        return {
            "status": "warning",
            "message": error_msg,
        }

    # 步驟 5：執行遷移
    try:
        migrated_data, migration_history = migrate_ticket(ticket_data)

        # 步驟 6：更新票據資料
        context["ticket_data"] = migrated_data

        # 步驟 7：記錄遷移結果
        migration_msg = f"Ticket {ticket_data.get('id')}: 已遷移 {detected_version} → {PROTOCOL_VERSION_CURRENT}"
        log_to_file(level="INFO", message=migration_msg, detail=migration_history)

        return {
            "status": "success",
            "reason": "migration_completed",
            "from_version": detected_version,
            "to_version": PROTOCOL_VERSION_CURRENT,
            "migration_history": migration_history,
        }

    except ProtocolVersionError as e:
        # 遷移失敗
        error_msg = f"Ticket {ticket_data.get('id')}: 遷移失敗 - {str(e)}"
        log_to_stderr(level="ERROR", message=error_msg)
        log_to_file(level="ERROR", message=error_msg, traceback=True)

        return {
            "status": "warning",
            "message": error_msg,
        }

# Hook 註冊設定（settings.json 中新增）
hook_config = {
    "name": "protocol-version-check-hook",
    "type": "PostToolUse",
    "trigger_on": ["ticket_load", "ticket_list", "ticket_validate"],
    "enabled": True,
    "priority": 100,  # 高優先級，在 Ticket 處理前執行
}
```

### 2.4 Dry-run 驗證腳本（虛擬碼）

```pseudocode
def run_migration_dry_run():
    """
    掃描所有現存 ticket，執行 dry-run 遷移驗證。

    Returns:
        dict，驗證報告 {
            "total": 總數,
            "successful": 成功數,
            "failed": 失敗數,
            "anomalies": [異常列表],
            "summary": 摘要,
        }
    """

    results = {
        "total": 0,
        "successful": 0,
        "failed": 0,
        "anomalies": [],
    }

    # 步驟 1：掃描 docs/work-logs/ 下所有 .md 檔案
    ticket_files = scan_all_ticket_files("docs/work-logs/")
    results["total"] = len(ticket_files)

    # 步驟 2：逐個驗證
    for ticket_file in ticket_files:
        try:
            # 讀取 ticket frontmatter
            ticket_data = read_ticket_frontmatter(ticket_file)

            # 執行 dry-run 遷移（不修改原檔案）
            migrated_data, migration_history = migrate_ticket(ticket_data)

            # 驗證遷移完整性
            if has_data_loss(ticket_data, migrated_data):
                results["anomalies"].append({
                    "file": ticket_file,
                    "type": "data_loss",
                    "message": "遷移過程中發生資訊遺失",
                })
                results["failed"] += 1
            else:
                results["successful"] += 1

        except Exception as e:
            results["anomalies"].append({
                "file": ticket_file,
                "type": "migration_error",
                "message": str(e),
            })
            results["failed"] += 1

    # 步驟 3：分類異常原因
    results["anomaly_summary"] = categorize_anomalies(results["anomalies"])

    # 步驟 4：產生驗證報告
    report = generate_dry_run_report(results)

    return results, report


def generate_dry_run_report(results):
    """
    產生人類可讀的驗證報告。

    報告格式：
    ```
    Protocol Version Migration Dry-Run Report
    ==========================================

    Summary:
    - Total Tickets: 42
    - Successful: 40
    - Failed: 0
    - Anomalies: 2

    Anomalies:
    - docs/work-logs/v0.1.0/tickets/0.1.0-W1-001.md
      Type: format_invalid
      Reason: 無效版本格式 "v2.0"

    - docs/work-logs/v0.1.1/tickets/0.1.1-W2-005.md
      Type: missing_field
      Reason: 缺少必填欄位 "type"

    Conclusion: 遷移成功率 95.2%（40/42），格式異常 ticket 需手動修正
    ```
    """

    success_rate = (results["successful"] / results["total"] * 100) if results["total"] > 0 else 0

    report = f"""
    Protocol Version Migration Dry-Run Report
    ==========================================

    Execution Time: {current_timestamp()}

    Summary:
    - Total Tickets: {results["total"]}
    - Successful: {results["successful"]}
    - Failed: {results["failed"]}
    - Anomalies: {len(results["anomalies"])}
    - Success Rate: {success_rate:.1f}%

    Anomaly Classification:
    {generate_anomaly_classification(results["anomaly_summary"])}

    Detailed Anomalies:
    {generate_anomaly_details(results["anomalies"])}

    Conclusion:
    {generate_conclusion(results)}
    """

    return report
```

### 3.1 Frontmatter Schema 變更

現有 Ticket 結構（前）：
```yaml
---
id: 0.1.0-W1-001
title: Task Title
type: IMP
status: pending
# ... 其他欄位
---
```

新增後（Phase 3b 完成後）：
```yaml
---
id: 0.1.0-W1-001
protocol_version: "2.0"  # 新欄位，位於 frontmatter 最上方
title: Task Title
type: IMP
status: pending
# ... 其他欄位
---
```

**重點設計決策**：
1. 版本欄位放在 frontmatter 開始位置（id 之後），便於快速偵測
2. 值為字串格式 "Major.Minor"（JSON-friendly，易於跨語言解析）
3. YAML 驗證規則：版本欄位為可選（舊 ticket 相容），新 ticket 必須包含

```
寫入 frontmatter protocol_version: "2.0"
    ↓
保存 Ticket 檔案
    ↓
Hook 在讀取時檢測版本 → 已是最新版本 → 無需遷移


載入舊 Ticket（無 protocol_version）
    ↓
Hook 偵測 protocol_version 缺失
    ↓
推斷版本為 PROTOCOL_VERSION_DEFAULT ("1.0")
    ↓
呼叫 migrate_ticket()
    ↓
查詢 PROTOCOL_VERSION_MIGRATIONS["1.0"]
    ↓
執行 migrate_v1_to_v2()（保留所有欄位，新增可選欄位預設值）
    ↓
更新版本為 "2.0"
    ↓
記錄遷移歷史
    ↓
返回遷移後的 Ticket 資料給消費端
```

## 5. 控制流程圖

```
ticket_load_operation
    ↓
protocol_version_check_hook triggered
    ├─ 讀取 protocol_version 欄位
    │   ├─ 欄位不存在 → 推斷為 "1.0"
    │   └─ 欄位存在 → 取得值
    ├─ 驗證版本格式
    │   ├─ 無效格式 → WARNING 日誌 + 返回警告 + 不阻止操作
    │   └─ 有效格式 → 繼續
    ├─ 比對版本
    │   ├─ version == PROTOCOL_VERSION_CURRENT → 無需遷移 → 返回成功
    │   ├─ version < PROTOCOL_VERSION_CURRENT → 執行遷移
    │   └─ version > PROTOCOL_VERSION_CURRENT → WARNING （未來版本）
    ├─ 執行遷移鏈
    │   ├─ 查詢遷移規則
    │   │   ├─ 規則存在 → 執行處理函式
    │   │   └─ 規則不存在 → ERROR 日誌 + 返回警告
    │   └─ 迴圈遷移至最新版本
    └─ 完成操作

dry_run_verification
    ├─ 掃描 docs/work-logs/ 所有 ticket 檔案
    ├─ 逐個執行 migrate_ticket() dry-run（不修改）
    ├─ 分類結果
    │   ├─ 成功 → 計數
    │   ├─ 失敗 → 記錄異常
    │   └─ 資訊遺失 → 記錄異常
    └─ 產生驗證報告（成功率、異常分類、詳細訊息）
```

## 7. 權宜方案與技術債務

### 7.1 最小可行實作

本 Phase 3a 規劃採用「最小化設計」：

| 項目 | 最小實作 | 未來可擴展 |
|------|---------|----------|
| 版本層級 | 兩層（Major.Minor）| 三層（Major.Minor.Patch）|
| 遷移鏈 | v1→v2 單一路徑 | 支援 v1→v2→v3→... 多級遷移 |
| Hook 觸發時機 | 被動檢測（讀取時） | 主動掃描+批量遷移 |
| 日誌記錄 | 基本錯誤訊息 | 詳細遷移步驟日誌 |

### 7.2 已知限制

| 限制 | 原因 | 影響 | 解決時機 |
|------|------|------|---------|
| 不支援多版本並存 | 簡化設計，不需要相容多個舊版本 | 大版本遷移時必須逐步執行 | Phase 4 或未來版本 |
| 不支援條件式遷移 | 初期所有舊 ticket 遷移到 v2.0 | 無法針對特定 ticket 子集進行有選擇性遷移 | 未來 ANA ticket 如有此需求再補 |
| 遷移結果不持久化 | Hook 執行時遷移，但不自動寫回 ticket 檔案 | 需要手動執行 dry-run 腳本或 Hook 觸發才能驗證 | Phase 4 可考慮自動持久化 |

### 7.3 技術債務標記

以下項目在 Phase 4 重構時應檢視：

- **TD-001**：遷移鏈硬編碼在 constants.py 中，未來應提取為配置檔（.yaml）
- **TD-002**：Hook 觸發時機為被動（讀取時），Phase 4 可評估主動掃描全量 ticket 的成本效益
- **TD-003**：日誌格式為純文字，未來應考慮結構化日誌（JSON）便於分析
- **TD-004**：Dry-run 驗證腳本為單次執行，未來可考慮定期自動執行和告警機制

| 檔案 | 調整項目 | 理由 |

## 8. 語言特定實作注意事項

### Python 考量

- **型別標註**：遷移函式應包含完整型別標註（Dict, List, Optional），便於靜態分析
- **異常処理**：定義 ProtocolVersionError 自訂異常類別，便於區分版本相關錯誤
- **正則表達式**：預編譯 PROTOCOL_VERSION_RE（使用 re.compile），避免重複編譯
- **日誌模組**：使用 logging 標準庫，配合 hook_utils.py 的日誌函式確保雙通道記錄
- **測試框架**：使用 pytest（同專案既有慣例），利用 parametrize 裝飾器覆蓋多版本測試場景

### 效能最佳化建議

- 版本格式驗證使用預編譯的正則表達式，避免每次都重新編譯
- 遷移鏈查詢使用字典存取，O(1) 時間複雜度
- dry-run 驗證可使用多執行緒並行掃描（版本 v0.1.3+ 考慮）

### 可能的技術挑戰

1. **YAML 解析**：frontmatter protocol_version 的 YAML 型別（字串 vs null vs 無欄位）需明確定義
2. **回溯相容性**：確保舊 ticket 讀取邏輯在無 protocol_version 欄位時不拋異常
3. **Hook 執行順序**：protocol-version-check-hook 應在其他 ticket 操作前執行，需檢視 settings.json 的優先級設定
